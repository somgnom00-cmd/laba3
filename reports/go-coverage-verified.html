type Array struct {
        data []string
}

func NewArray() *Array                               { return &Array{} }
func (a *Array) Type() string                        { return "Array" }
func (a *Array) Size() int                           { return len(a.data) }
func (a *Array) Values() []string                    { return append([]string(nil), a.data...) }
func (a *Array) Clear()                              { a.data = nil }
func (a *Array) PushBack(value string)               { a.data = append(a.data, value) }
func (a *Array) appendSerialized(value string) error { a.PushBack(value); return nil }

func (a *Array) Insert(index int, value string) error {
        if index < 0 || index > len(a.data) {
                return fmt.Errorf("array index out of range")
        }
        a.data = append(a.data, "")
        copy(a.data[index+1:], a.data[index:])
        a.data[index] = value
        return nil
}

func (a *Array) Erase(index int) error {
        if index < 0 || index >= len(a.data) {
                return fmt.Errorf("array index out of range")
        }
        copy(a.data[index:], a.data[index+1:])
        a.data = a.data[:len(a.data)-1]
        return nil
}

func (a *Array) At(index int) (string, error) {
        if index < 0 || index >= len(a.data) {
                return "", fmt.Errorf("array index out of range")
        }
        return a.data[index], nil
}

func (a *Array) Find(value string) int {
        for i, stored := range a.data {
                if stored == value {
                        return i
                }
        }
        return -1
}

type treeNode struct {
        value       string
        left, right *treeNode
}

type CompleteBinaryTree struct {
        root *treeNode
        size int
}

func NewCompleteBinaryTree() *CompleteBinaryTree                  { return &CompleteBinaryTree{} }
func (t *CompleteBinaryTree) Type() string                        { return "CompleteBinaryTree" }
func (t *CompleteBinaryTree) Size() int                           { return t.size }
func (t *CompleteBinaryTree) Clear()                              { t.root, t.size = nil, 0 }
func (t *CompleteBinaryTree) appendSerialized(value string) error { t.Insert(value); return nil }

func (t *CompleteBinaryTree) Insert(value string) {
        added := &treeNode{value: value}
        if t.root == nil {
                t.root = added
                t.size = 1
                return
        }
        pending := []*treeNode{t.root}
        for len(pending) > 0 {
                current := pending[0]
                pending = pending[1:]
                if current.left == nil {
                        current.left = added
                        t.size++
                        return
                }
                if current.right == nil {
                        current.right = added
                        t.size++
                        return
                }
                pending = append(pending, current.left, current.right)
        }
}

func (t *CompleteBinaryTree) Values() []string {
        result := make([]string, 0, t.size)
        if t.root == nil {
                return result
        }
        pending := []*treeNode{t.root}
        for len(pending) > 0 {
                current := pending[0]
                pending = pending[1:]
                result = append(result, current.value)
                if current.left != nil {
                        pending = append(pending, current.left)
                }
                if current.right != nil {
                        pending = append(pending, current.right)
                }
        }
        return result
}

func (t *CompleteBinaryTree) Contains(value string) bool {
        for _, stored := range t.Values() {
                if stored == value {
                        return true
                }
        }
        return false
}

func (t *CompleteBinaryTree) IsComplete() bool {
        if t.root == nil {
                return true
        }
        pending := []*treeNode{t.root}
        gap := false
        for len(pending) > 0 {
                current := pending[0]
                pending = pending[1:]
                if current == nil {
                        gap = true
                        continue
                }
                if gap {
                        return false
                }
                pending = append(pending, current.left, current.right)
        }
        return true
}

import "fmt"

type doubleNode struct {
        value string
        prev  *doubleNode
        next  *doubleNode
}

type DoublyList struct {
        head *doubleNode
        tail *doubleNode
        size int
}

func NewDoublyList() *DoublyList                          { return &DoublyList{} }
func (l *DoublyList) Type() string                        { return "DoublyList" }
func (l *DoublyList) Size() int                           { return l.size }
func (l *DoublyList) Clear()                              { l.head, l.tail, l.size = nil, nil, 0 }
func (l *DoublyList) appendSerialized(value string) error { l.PushBack(value); return nil }

func (l *DoublyList) PushFront(value string) {
        added := &doubleNode{value: value, next: l.head}
        if l.head != nil {
                l.head.prev = added
        } else {
                l.tail = added
        }
        l.head = added
        l.size++
}

func (l *DoublyList) PushBack(value string) {
        added := &doubleNode{value: value, prev: l.tail}
        if l.tail != nil {
                l.tail.next = added
        } else {
                l.head = added
        }
        l.tail = added
        l.size++
}

func (l *DoublyList) nodeAt(index int) (*doubleNode, error) {
        if index < 0 || index >= l.size {
                return nil, fmt.Errorf("list index out of range")
        }
        if index < l.size/2 {
                node := l.head
                for i := 0; i < index; i++ {
                        node = node.next
                }
                return node, nil
        }
        node := l.tail
        for i := l.size - 1; i > index; i-- {
                node = node.prev
        }
        return node, nil
}

func (l *DoublyList) Insert(index int, value string) error {
        if index < 0 || index > l.size {
                return fmt.Errorf("list index out of range")
        }
        if index == 0 {
                l.PushFront(value)
                return nil
        }
        if index == l.size {
                l.PushBack(value)
                return nil
        }
        current, _ := l.nodeAt(index)
        added := &doubleNode{value: value, prev: current.prev, next: current}
        current.prev.next = added
        current.prev = added
        l.size++
        return nil
}

func (l *DoublyList) Erase(index int) error {
        node, err := l.nodeAt(index)
        if err != nil {
                return err
        }
        if node.prev != nil {
                node.prev.next = node.next
        } else {
                l.head = node.next
        }
        if node.next != nil {
                node.next.prev = node.prev
        } else {
                l.tail = node.prev
        }
        l.size--
        return nil
}

func (l *DoublyList) Values() []string {
        result := make([]string, 0, l.size)
        for node := l.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

func (l *DoublyList) ReverseValues() []string {
        result := make([]string, 0, l.size)
        for node := l.tail; node != nil; node = node.prev {
                result = append(result, node.value)
        }
        return result
}

import (
        "fmt"
        "strconv"
        "strings"
)

type cellState uint8

const (
        empty cellState = iota
        occupied
        deleted
)

type hashCell struct {
        key, value int
        state      cellState
}

type OpenHashTable struct {
        cells      []hashCell
        size       int
        deleted    int
        doubleHash bool
}

func NewLinearHashTable(capacity int) *OpenHashTable {
        return newOpenHashTable(capacity, false)
}

func NewDoubleHashTable(capacity int) *OpenHashTable {
        return newOpenHashTable(capacity, true)
}

func newOpenHashTable(capacity int, doubleHash bool) *OpenHashTable {
        if capacity < 3 {
                capacity = 3
        }
        return &OpenHashTable{cells: make([]hashCell, nextPrime(capacity)), doubleHash: doubleHash}
}

func (h *OpenHashTable) Type() string {
        if h.doubleHash {
                return "DoubleHashTable"
        }
        return "LinearHashTable"
}

func (h *OpenHashTable) Size() int { return h.size }

func (h *OpenHashTable) Clear() {
        h.cells = make([]hashCell, len(h.cells))
        h.size, h.deleted = 0, 0
}

func magnitude(key int) uint {
        if key < 0 {
                return uint(-(key + 1)) + 1
        }
        return uint(key)
}

func foldKey(key int) int {
        number := magnitude(key)
        var sum uint
        for {
                sum += number % 1000
                number /= 1000
                if number == 0 {
                        return int(sum)
                }
        }
}

func nextPrime(value int) int {
        for {
                prime := value >= 2
                for divisor := 2; prime && divisor <= value/divisor; divisor++ {
                        if value%divisor == 0 {
                                prime = false
                        }
                }
                if prime {
                        return value
                }
                value++
        }
}

func (h *OpenHashTable) step(key int) int {
        if !h.doubleHash {
                return 1
        }
        number := magnitude(key)
        return 1 + int(number%uint(len(h.cells)-1))
}

func (h *OpenHashTable) locate(key int) int {
        if len(h.cells) == 0 {
                return -1
        }
        start := foldKey(key) % len(h.cells)
        step := h.step(key)
        for attempt := 0; attempt < len(h.cells); attempt++ {
                index := (start + attempt*step) % len(h.cells)
                if h.cells[index].state == empty {
                        return -1
                }
                if h.cells[index].state == occupied && h.cells[index].key == key {
                        return index
                }
        }
        return -1
}

func (h *OpenHashTable) Set(key, value int) {
        if len(h.cells) == 0 {
                h.cells = make([]hashCell, 17)
        }
        if index := h.locate(key); index >= 0 {
                h.cells[index].value = value
                return
        }
        if (h.size+h.deleted+1)*10 > len(h.cells)*7 {
                capacity := len(h.cells)
                if (h.size+1)*10 > capacity*7 {
                        capacity = capacity*2 + 1
                }
                h.rebuild(capacity)
        }
        start, step := foldKey(key)%len(h.cells), h.step(key)
        free := -1
        for attempt := 0; attempt < len(h.cells); attempt++ {
                index := (start + attempt*step) % len(h.cells)
                if h.cells[index].state == deleted && free < 0 {
                        free = index
                }
                if h.cells[index].state == empty {
                        if free < 0 {
                                free = index
                        }
                        break
                }
        }
        if h.cells[free].state == deleted {
                h.deleted--
        }
        h.cells[free] = hashCell{key: key, value: value, state: occupied}
        h.size++
}

func (h *OpenHashTable) rebuild(capacity int) {
        old := h.cells
        h.cells = make([]hashCell, nextPrime(capacity))
        h.size, h.deleted = 0, 0
        for _, cell := range old {
                if cell.state == occupied {
                        h.Set(cell.key, cell.value)
                }
        }
}

func (h *OpenHashTable) Get(key int) (int, bool) {
        index := h.locate(key)
        if index < 0 {
                return 0, false
        }
        return h.cells[index].value, true
}

func (h *OpenHashTable) Delete(key int) bool {
        index := h.locate(key)
        if index < 0 {
                return false
        }
        h.cells[index].state = deleted
        h.size--
        h.deleted++
        return true
}

func (h *OpenHashTable) Values() []string {
        result := make([]string, 0, h.size)
        for _, cell := range h.cells {
                if cell.state == occupied {
                        result = append(result, fmt.Sprintf("%d=%d", cell.key, cell.value))
                }
        }
        return result
}

func (h *OpenHashTable) appendSerialized(value string) error {
        parts := strings.SplitN(value, "=", 2)
        if len(parts) != 2 {
                return fmt.Errorf("invalid hash table record")
        }
        key, err := strconv.Atoi(parts[0])
        if err != nil {
                return err
        }
        stored, err := strconv.Atoi(parts[1])
        if err != nil {
                return err
        }
        h.Set(key, stored)
        return nil
}

type Queue struct {
        head *singleNode
        tail *singleNode
        size int
}

func NewQueue() *Queue                               { return &Queue{} }
func (q *Queue) Type() string                        { return "Queue" }
func (q *Queue) Size() int                           { return q.size }
func (q *Queue) Clear()                              { q.head, q.tail, q.size = nil, nil, 0 }
func (q *Queue) appendSerialized(value string) error { q.Push(value); return nil }

func (q *Queue) Push(value string) {
        added := &singleNode{value: value}
        if q.tail != nil {
                q.tail.next = added
        } else {
                q.head = added
        }
        q.tail = added
        q.size++
}

func (q *Queue) Front() (string, error) {
        if q.head == nil {
                return "", fmt.Errorf("queue is empty")
        }
        return q.head.value, nil
}

func (q *Queue) Pop() (string, error) {
        value, err := q.Front()
        if err != nil {
                return "", err
        }
        q.head = q.head.next
        q.size--
        if q.head == nil {
                q.tail = nil
        }
        return value, nil
}

func (q *Queue) Values() []string {
        result := make([]string, 0, q.size)
        for node := q.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

import (
        "encoding/gob"
        "encoding/json"
        "fmt"
        "os"
)

type storedContainer struct {
        Type   string   `json:"type"`
        Values []string `json:"values"`
}

func SaveText(container Container, file string) (err error) {
        output, err := os.Create(file)
        if err != nil {
                return err
        }
        defer func() {
                if closeErr := output.Close(); err == nil {
                        err = closeErr
                }
        }()
        encoder := json.NewEncoder(output)
        encoder.SetIndent("", "  ")
        return encoder.Encode(storedContainer{Type: container.Type(), Values: container.Values()})
}

func LoadText(container Container, file string) error {
        input, err := os.Open(file)
        if err != nil {
                return err
        }
        defer func() { _ = input.Close() }()
        var stored storedContainer
        if err := json.NewDecoder(input).Decode(&stored); err != nil {
                return err
        }
        return restore(container, stored)
}

func SaveBinary(container Container, file string) (err error) {
        output, err := os.Create(file)
        if err != nil {
                return err
        }
        defer func() {
                if closeErr := output.Close(); err == nil {
                        err = closeErr
                }
        }()
        return gob.NewEncoder(output).Encode(storedContainer{Type: container.Type(), Values: container.Values()})
}

func LoadBinary(container Container, file string) error {
        input, err := os.Open(file)
        if err != nil {
                return err
        }
        defer func() { _ = input.Close() }()
        var stored storedContainer
        if err := gob.NewDecoder(input).Decode(&stored); err != nil {
                return err
        }
        return restore(container, stored)
}

func restore(container Container, stored storedContainer) error {
        if stored.Type != container.Type() {
                return fmt.Errorf("container type mismatch: %s and %s", stored.Type, container.Type())
        }
        // Заполняем временный контейнер. При ошибке прежние данные не меняются.
        switch destination := container.(type) {
        case *Array:
                next := NewArray()
                return replace(stored, next, func() { *destination = *next })
        case *SinglyList:
                next := NewSinglyList()
                return replace(stored, next, func() { *destination = *next })
        case *DoublyList:
                next := NewDoublyList()
                return replace(stored, next, func() { *destination = *next })
        case *Stack:
                next := NewStack()
                return replace(stored, next, func() { *destination = *next })
        case *Queue:
                next := NewQueue()
                return replace(stored, next, func() { *destination = *next })
        case *CompleteBinaryTree:
                next := NewCompleteBinaryTree()
                return replace(stored, next, func() { *destination = *next })
        case *OpenHashTable:
                next := newOpenHashTable(len(destination.cells), destination.doubleHash)
                return replace(stored, next, func() { *destination = *next })
        default:
                return fmt.Errorf("container does not support deserialization")
        }
}

func replace(stored storedContainer, target serializable, commit func()) error {
        for _, value := range stored.Values {
                if err := target.appendSerialized(value); err != nil {
                        return err
                }
        }
        commit()
        return nil
}

type SinglyList struct {

        tail *singleNode
        size int

func NewSinglyList() *SinglyList                          { return &SinglyList{} }
func (l *SinglyList) Type() string                        { return "SinglyList" }
func (l *SinglyList) Size() int                           { return l.size }
func (l *SinglyList) Clear()                              { l.head, l.tail, l.size = nil, nil, 0 }
func (l *SinglyList) appendSerialized(value string) error { l.PushBack(value); return nil }

func (l *SinglyList) PushFront(value string) {
        added := &singleNode{value: value, next: l.head}
        l.head = added
        if l.tail == nil {
                l.tail = added
        }
        l.size++
}

func (l *SinglyList) PushBack(value string) {
        added := &singleNode{value: value}
        if l.tail != nil {
                l.tail.next = added
        } else {
                l.head = added
        }
        l.tail = added
        l.size++

func (l *SinglyList) nodeAt(index int) (*singleNode, error) {
        if index < 0 || index >= l.size {
                return nil, fmt.Errorf("list index out of range")
        }
        node := l.head
        for i := 0; i < index; i++ {
                node = node.next
        }
        return node, nil
}

func (l *SinglyList) Insert(index int, value string) error {
        if index < 0 || index > l.size {
                return fmt.Errorf("list index out of range")
        }
        if index == 0 {
                l.PushFront(value)
                return nil
        }
        if index == l.size {
                l.PushBack(value)
                return nil
        }
        before, _ := l.nodeAt(index - 1)
        before.next = &singleNode{value: value, next: before.next}
        l.size++
        return nil
}

func (l *SinglyList) Erase(index int) error {
        if index < 0 || index >= l.size {
                return fmt.Errorf("list index out of range")
        }
        if index == 0 {
                l.head = l.head.next
                l.size--
                if l.size == 0 {
                        l.tail = nil
                }
                return nil
        }
        before, _ := l.nodeAt(index - 1)
        if before.next == l.tail {
                l.tail = before
        }
        before.next = before.next.next
        l.size--
        return nil
}

func (l *SinglyList) Values() []string {
        result := make([]string, 0, l.size)
        for node := l.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

package structures

import "fmt"

type Stack struct {
        top  *singleNode
        size int
}

func NewStack() *Stack             { return &Stack{} }
func (s *Stack) Type() string      { return "Stack" }
func (s *Stack) Size() int         { return s.size }
func (s *Stack) Clear()            { s.top, s.size = nil, 0 }
func (s *Stack) Push(value string) { s.top = &singleNode{value: value, next: s.top}; s.size++ }

func (s *Stack) appendSerialized(value string) error {
        added := &singleNode{value: value}
        if s.top == nil {
                s.top = added
        } else {
                bottom := s.top
                for bottom.next != nil {
                        bottom = bottom.next
                }
                bottom.next = added
        }
        s.size++
        return nil
}

func (s *Stack) Top() (string, error) {
        if s.top == nil {
                return "", fmt.Errorf("stack is empty")
        }
        return s.top.value, nil
}

func (s *Stack) Pop() (string, error) {
        value, err := s.Top()
        if err != nil {
                return "", err
        }
        s.top = s.top.next
        s.size--
        return value, nil
}

func (s *Stack) Values() []string {
        result := make([]string, 0, s.size)
        for node := s.top; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}
