  24     const auto prime = [](std::size_t number) {

  26             return false;

 426         result.push_back(node->value);
 427     return result;
 428 }

 430 void Stack::appendSerialized(const std::string &value) {
 431     Node *added = new Node{value};
 432     if (!top_) {
 433         top_ = added;
 434     } else {
 435         Node *bottom = top_;
 436         while (bottom->next)
 437             bottom = bottom->next;
 438         bottom->next = added;
 439     }
 440     ++size_;
 441 }

 444     while (top_) {
 445         Node *removed = top_;
 446         top_ = top_->next;
 447         delete removed;
 448     }
 449     size_ = 0;
 450 }

 493 std::string Queue::pop() {
 494     if (!head_)
 495         throw std::out_of_range("очередь пуста");
 496     Node *removed = head_;
 497     std::string value = std::move(removed->value);
 498     head_ = removed->next;
 499     if (!head_)
 500         tail_ = nullptr;
 501     delete removed;
 502     --size_;
 503     return value;
 504 }

 560 void CompleteBinaryTree::insert(const std::string &value) {
 561     auto owner = std::make_unique<Node>();
 562     owner->value = value;
 563     if (!root_) {
 564         root_ = owner.release();
 565         ++size_;
 566         return;
 567     }
 568     std::queue<Node *> pending;
 569     pending.push(root_);
 570     while (!pending.empty()) {
 571         Node *current = pending.front();
 572         pending.pop();
 573         if (!current->left) {
 574             current->left = owner.release();
 575             ++size_;
 576             return;
 577         }
 578         if (!current->right) {
 579             current->right = owner.release();
 580             ++size_;
 581             return;
 582         }
 583         pending.push(current->left);
 584         pending.push(current->right);
 585     }
 586 }

 588 bool CompleteBinaryTree::contains(const std::string &value) const {
 589     const auto stored = values();
 590     return std::find(stored.begin(), stored.end(), value) != stored.end();

 593 bool CompleteBinaryTree::isComplete() const {
 594     if (!root_)
 595         return true;
 596     std::queue<Node *> pending;
 597     pending.push(root_);
 598     bool gap = false;
 599     while (!pending.empty()) {
 600         Node *current = pending.front();
 601         pending.pop();
 602         if (!current) {
 603             gap = true;
 604         } else {
 605             if (gap)
 606                 return false;
 607             pending.push(current->left);
 608             pending.push(current->right);
 609         }
 610     }
 611     return true;
 612 }

 614 std::vector<std::string> CompleteBinaryTree::values() const {
 615     std::vector<std::string> result;
 616     result.reserve(size_);
 617     if (!root_)
 618         return result;
 619     std::queue<Node *> pending;
 620     pending.push(root_);
 621     while (!pending.empty()) {
 622         Node *current = pending.front();
 623         pending.pop();
 624         result.push_back(current->value);
 625         if (current->left)
 626             pending.push(current->left);
 627         if (current->right)
 628             pending.push(current->right);
 629     }
 630     return result;
 631 }
